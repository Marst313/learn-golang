import "fmt"

type Person struct {
	firstName     string
	lastName      string
	age           int
	address       Address

}

type PhoneHomeCell struct {
	home string
	cell string
}

type Address struct {
	city    string
	country string
}

func (a Person) fullName() string {
	return a.firstName + " " + a.lastName
}

func (p *Person) incrementAgeByOne() {
	p.age += 1
}

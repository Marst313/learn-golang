	salt, err := generateSalt()

	if err != nil {
		fmt.Println("Error generating salt ", err)

	// Hash the password with salt
	signUpHash := hashPassword(password, salt)

	// Store password in db, just printing now

	fmt.Println("Hash", signUpHash) // simulate storing in db

func generateSalt() ([]byte, error) {
	salt := make([]byte, 16)

	_, err := io.ReadFull(rand.Reader, salt)
	if err != nil {
		return nil, err
	}

	return salt, nil
}

// FUNCTION TO HASH PASSWORD WITH SALT
func hashPassword(password string, salt []byte) string {
	saltedPassword := append(salt, []byte(password)...)
	hash256 := sha256.Sum256(saltedPassword)

	return base64.StdEncoding.EncodeToString(hash256[:])
}

package main

import (
	"fmt"
	"math"
)

// If use capital Geometry it can be access from others file || Exported
type geometry interface {
	area() float64
	perim() float64
}

// ? Type 1
type circle struct {
	radius float64
}

// ! #1 Method area circle
func (c circle) area() float64 {
	return math.Pi * c.radius * c.radius
}

// ! #2 Method perim circle
func (c circle) perim() float64 {
	return 2 * math.Pi * c.radius
}

// ! #3 Method diameter circle
func (c circle) diameter() float64 {
	return 2 * c.radius
}

// ? Type 2
type rect struct {
	width, height float64
}

// ! #1 Method area rectangle
func (r rect) area() float64 {
	return math.Pi * r.height * r.width
}

// ! #2 Method perim rectangle
func (r rect) perim() float64 {
	return 2 * (r.height + r.width)
}

func printType(i interface{}) {
	switch i.(type) {
	case int:
		fmt.Println("Type : Int")
	case string:
		fmt.Println("Type : String")
	default:
		fmt.Println("Type : Unknown")
	}
}

func myPrinter(i ...interface{}) {
	for _, v := range i {
		fmt.Println(v)
	}
}

package main

import "fmt"

type Rectangle struct {
	length float64
	width  float64
}

type MyInt int

type Shape struct {
	Rectangle
}

// ! #1 Method with value receiver
func (r Rectangle) Area() float64 {
	return r.length * r.width
}

// ! #2 Method with pointer receiver
func (r *Rectangle) Scale(factor float64) {
	r.length *= factor // shorthand  ( r.length = r.length * factor)
	r.width *= factor
}

// ! #3 Method on a user-defined type
func (m MyInt) isPositive() bool {
	return m > 0
}

func (MyInt) welcomeMessage() string {
	return "Welcome to MyInte Type"
}

func main() {

	// ? Declare New Instance
	rect := Rectangle{length: 10, width: 9}
	area := rect.Area()
	fmt.Println("Area of rectangle with width 9 and length 10 is", area)

	const factor float64 = 2

	// ? Scale
	rect.Scale(float64(factor))
	area = rect.Area()
	fmt.Printf("Area of rectangle with a factor of %v is %v\n", factor, area)

	// ? Declare New Instance for MyInt
	num := MyInt(-5)
	num1 := MyInt(9)
	fmt.Println(num.isPositive())
	fmt.Println(num1.isPositive())
	fmt.Println(num.welcomeMessage())

	// ? Declare New Instance for Shape
	s := Shape{Rectangle: Rectangle{length: 10, width: 9}}
	fmt.Println(s.Area())
	fmt.Println(s.Rectangle.Area())

}

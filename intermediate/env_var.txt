	fmt.Println("FRUIT env var:", os.Getenv("FRUIT"))

		return
	}
	fmt.Println("Unset env var done on key FRUIT")
	fmt.Println("FRUIT env var:", os.Getenv("FRUIT"))
	fmt.Println("---------------------------------------------")

	str := "a=b=c=d=e"
	// n = 1 return "a=b=c=d"

	fmt.Println(strings.SplitN(str, "=", 3))
